{{- /* Link render hook — mirrors the Astro theme's `rehypeExternalLinks` pass: external links get `rel="nofollow noopener noreferrer"`, `target="_blank"`, and a trailing ` ↗` hint wrapped in `user-select:none` so it is not copied. The arrow's text and style come from `params.content.externalLinks`. */ -}} {{- $dest := .Destination -}} {{- $external := or (hasPrefix $dest "http://") (hasPrefix $dest "https://") (hasPrefix $dest "//") -}} {{- $cfg := site.Params.content.externalLinks -}} {{ .Text | safeHTML }}{{ if $external }}{{ $cfg.content | default " ↗" }}{{ end }}
